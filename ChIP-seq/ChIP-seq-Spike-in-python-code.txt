#!/usr/bin/env bash
set -euo pipefail


THREADS=8
PREFIX="sample1" # Base name for output
R1_RAW="raw_reads/${PREFIX}_R1.fastq.gz"
R2_RAW="raw_reads/${PREFIX}_R2.fastq.gz"


REF_POMBE="refs/pombe/ASM294v3.fa"
REF_CEREVISIAE="refs/cerevisiae/R64.fa"
BLACKLIST_POMBE="refs/pombe/pombase_blacklist.bed"


INPUT_BAM="bams/input_markdup.bam"
INPUT_SPIKE_SCALED_BW="tracks/input_spike_norm.bw"

mkdir -p qc trimmed alignments bams tracks peaks metagene


echo "==> Running QC and Trimming..."
fastqc -t ${THREADS} -o qc/ ${R1_RAW} ${R2_RAW}
multiqc qc/ -o qc/multiqc_pretrim

trim_galore --paired --phred20 -o trimmed/ ${R1_RAW} ${R2_RAW}

R1_TRIM="trimmed/${PREFIX}_R1_val_1.fq.gz"
R2_TRIM="trimmed/${PREFIX}_R2_val_2.fq.gz"


echo "==> Aligning to S. pombe reference..."
bwa mem -t ${THREADS} ${REF_POMBE} ${R1_TRIM} ${R2_TRIM} | \
  samtools view -@ ${THREADS} -u - | \
  samtools sort -@ ${THREADS} -n -o alignments/${PREFIX}_pombe_namesort.bam -

samtools fixmate -@ ${THREADS} -m alignments/${PREFIX}_pombe_namesort.bam alignments/${PREFIX}_pombe_fixmate.bam
samtools sort -@ ${THREADS} -o alignments/${PREFIX}_pombe_coordsort.bam alignments/${PREFIX}_pombe_fixmate.bam
samtools markdup -@ ${THREADS} alignments/${PREFIX}_pombe_coordsort.bam bams/${PREFIX}_pombe_markdup.bam
samtools index bams/${PREFIX}_pombe_markdup.bam

rm alignments/${PREFIX}_pombe_namesort.bam alignments/${PREFIX}_pombe_fixmate.bam alignments/${PREFIX}_pombe_coordsort.bam


echo "==> Aligning to S. cerevisiae spike-in reference..."
bwa mem -t ${THREADS} ${REF_CEREVISIAE} ${R1_TRIM} ${R2_TRIM} | \
  samtools view -@ ${THREADS} -b -o alignments/${PREFIX}_cerevisiae.bam -

SPIKE_READS=$(samtools view -c -F 4 alignments/${PREFIX}_cerevisiae.bam)
echo "Spike-in mapped reads: ${SPIKE_READS}"

SCALE_FACTOR=$(awk -v reads="${SPIKE_READS}" 'BEGIN { printf "%.8f", 1000000 / reads }')
echo "Spike-in Scale Factor: ${SCALE_FACTOR}"


echo "==> Generating Spike-in normalized coverage track (deepTools)..."
bamCoverage \
  -b bams/${PREFIX}_pombe_markdup.bam \
  -o tracks/${PREFIX}_spike_norm.bw \
  --binSize 10 \
  --scaleFactor ${SCALE_FACTOR} \
  -p ${THREADS}

echo "==> Generating RPM-normalized coverage track (BEDTools + UCSC)..."
PROPER_READS=$(samtools view -c -F 260 bams/${PREFIX}_pombe_markdup.bam)
RPM_SCALE=$(awk -v reads="${PROPER_READS}" 'BEGIN { printf "%.8f", 1000000 / reads }')


samtools idxstats bams/${PREFIX}_pombe_markdup.bam | cut -f1,2 > refs/pombe/chrom.sizes

bedtools genomecov -ibam bams/${PREFIX}_pombe_markdup.bam -bg -scale ${RPM_SCALE} > tracks/${PREFIX}_rpm.bedgraph
bedGraphToBigWig tracks/${PREFIX}_rpm.bedgraph refs/pombe/chrom.sizes tracks/${PREFIX}_rpm.bw
rm tracks/${PREFIX}_rpm.bedgraph

echo "==> Calling peaks with MACS2..."

macs2 callpeak \
  -t bams/${PREFIX}_pombe_markdup.bam \
  -c ${INPUT_BAM} \
  -f BAMPE \
  -g 1.25e7 \
  -q 0.05 \
  -n ${PREFIX} \
  --outdir peaks/

bedtools intersect -v -a peaks/${PREFIX}_peaks.narrowPeak -b ${BLACKLIST_POMBE} | \
  awk '$7 > 2' > peaks/${PREFIX}_filtered_enrichment_gt2.bed

echo "==> Computing Relative Enrichment Ratio (ChIP / Input)..."
bigwigCompare \
  -b1 tracks/${PREFIX}_spike_norm.bw \
  -b2 ${INPUT_SPIKE_SCALED_BW} \
  -o tracks/${PREFIX}_rER.bw \
  --operation ratio \
  --binSize 10 \
  --smoothLength 100 \
  -p ${THREADS}

# Optional: Background subtraction (e.g., subtracting NLS-GFP control track)
# bigwigCompare \
#   -b1 tracks/${PREFIX}_spike_norm.bw \
#   -b2 tracks/NLS_GFP_spike_norm.bw \
#   -o tracks/${PREFIX}_subtracted.bw \
#   --operation subtract \
#   --binSize 10 \
#   -p ${THREADS}

echo "==> Generating metagene matrices and profiles..."
GENES_BED="refs/pombe/pombe_genes.bed" # BED file containing annotations (or ranked subsets)

computeMatrix scale-regions \
  -S tracks/${PREFIX}_spike_norm.bw \
  -R ${GENES_BED} \
  --beforeRegionStartLength 1000 \
  --regionBodyLength 3000 \
  --afterRegionStartLength 1500 \
  --binSize 10 \
  -o metagene/${PREFIX}_matrix.gz \
  -p ${THREADS}

plotProfile \
  -m metagene/${PREFIX}_matrix.gz \
  -out metagene/${PREFIX}_profile.pdf \
  --plotType lines \
  --perGroup

echo "==> Pipeline complete! Output tracks and peak BEDs are ready for IGV visualization."
